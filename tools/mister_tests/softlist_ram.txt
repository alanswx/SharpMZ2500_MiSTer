# MZ-2000 RAM images (MAME 0.289 mz2000_snap) through the direct loader (OSD "Load RAM image", ioctl index 2).
# Boot mode left at MZ-2500 on purpose: the loader switches to MZ-2000 itself. Start keys from the title screens,
# then a little play input (cursor keys and the ten-key pad, as these games use one or the other).
cfg bootmode 2500
cfg keyboard jp
load SL/ram_binary
wait 8
shot ram_binary_title
key SPACE 0.2
wait 3
shot ram_binary_k0
key RIGHT 1.0
key KP6 1.0
key UP 0.6
key KP8 0.6
key LEFT 1.0
key KP4 1.0
key SPACE 0.2
wait 1
shot ram_binary_play
wait 6
shot ram_binary_play2
